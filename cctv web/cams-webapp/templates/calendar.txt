    <script>
      // Initialize Lucide icons
      lucide.createIcons();

      // Original JS starts here
      function fmtDate(d) {
        return d.toISOString().slice(0,10);
      }
      function toEpoch(d) {
        return Math.floor(d.getTime() / 1000);
      }
      function startOfDay(d) {
        const x = new Date(d);
        x.setHours(0,0,0,0);
        return x;
      }
      function endOfDay(d) {
        const x = new Date(d);
        x.setHours(23,59,59,999);
        return x;
      }

      function defaultRange() {
        const to = new Date();
        const from = new Date();
        from.setDate(from.getDate() - 6);
        return { from, to };
      }

      function setRangeInputs(from, to) {
        const f = document.getElementById('from-date');
        const t = document.getElementById('to-date');
        f.value = fmtDate(from);
        t.value = fmtDate(to);
        const rangeText = document.getElementById('range-text');
        const fromStr = from.toLocaleDateString(undefined, { month: 'short', day: 'numeric', year: 'numeric' });
        const toStr = to.toLocaleDateString(undefined, { month: 'short', day: 'numeric', year: 'numeric' });
        rangeText.textContent = `${fromStr} ? ${toStr}`;
      }

      function applyPreset(days) {
        const to = new Date();
        const from = new Date();
        from.setDate(from.getDate() - (days - 1));
        return { from, to };
      }

      async function loadNvrs() {
        const res = await fetch('/api/nvrs');
        return await res.json();
      }

      async function loadEvents(from, to) {
        const q = new URLSearchParams({ from: String(toEpoch(from)), to: String(toEpoch(to)) });
        const res = await fetch(`/api/events?${q.toString()}`);
        return await res.json();
      }

      function buildDays(from, to) {
        const days = [];
        const cur = new Date(startOfDay(from));
        const end = endOfDay(to);
        while (cur <= end) {
          days.push(new Date(cur));
          cur.setDate(cur.getDate() + 1);
        }
        return days;
      }

      function computeDailyDowntime(dayStart, dayEnd, offlineIntervals) {
        let total = 0;
        for (const iv of offlineIntervals) {
          const s = new Date(iv.start * 1000);
          const e = new Date((iv.end ?? Math.floor(Date.now()/1000)) * 1000);
          const overlapStart = new Date(Math.max(s.getTime(), dayStart.getTime()));
          const overlapEnd = new Date(Math.min(e.getTime(), dayEnd.getTime()));
          const durMs = overlapEnd - overlapStart;
          if (durMs > 0) total += durMs;
        }
        const dayMs = dayEnd - dayStart;
        const ratio = dayMs > 0 ? total / dayMs : 0;
        return { millis: total, ratio };
      }

      function colorForRatio(r) {
        const c = Math.max(0, Math.min(1, r));
        const hue = 120 - Math.round(120 * c);
        return `hsl(${hue}, 90%, 45%)`;
      }

      function renderCalendar(nvrs, events, days) {
        const container = document.getElementById('calendar-container');
        container.innerHTML = '';
        const wrapper = document.createElement('div');
        wrapper.className = 'calendar-wrapper';

        const headerRow = document.createElement('div');
        headerRow.className = 'calendar-row header';
        headerRow.style.gridTemplateColumns = `220px repeat(${days.length}, 1fr)`;
        const nameHeader = document.createElement('div');
        nameHeader.className = 'cell name';
        nameHeader.textContent = 'NVR (uptime %)';
        headerRow.appendChild(nameHeader);
        const headerDayCells = [];
        for (let i = 0; i < days.length; i++) {
          const d = days[i];
          const c = document.createElement('div');
          c.className = 'cell day';
          const wk = d.toLocaleDateString(undefined, { weekday: 'short' });
          const dt = d.toLocaleDateString();
          c.textContent = `${wk}\n${dt}`;
          headerRow.appendChild(c);
          headerDayCells.push(c);
        }
        wrapper.appendChild(headerRow);

        const filter = (document.getElementById('nvr-filter').value || '').toLowerCase();
        const filteredNvrs = nvrs.filter(n => {
          const txt = ((n.name || '') + ' ' + (n.ip || '')).toLowerCase();
          return !filter || txt.includes(filter);
        });
        for (const nvr of filteredNvrs) {
          const row = document.createElement('div');
          row.className = 'calendar-row';
          row.style.gridTemplateColumns = `220px repeat(${days.length}, 1fr)`;
          const nameCell = document.createElement('div');
          nameCell.className = 'cell name';
          const offline = (events[nvr.ip] || []).filter(x => x.type === 'offline');
          const totalMs = (endOfDay(days[days.length-1]) - startOfDay(days[0]));
          let totalDownMs = 0;
          for (let j = 0; j < days.length; j++) {
            const d = days[j];
            const dayStart = startOfDay(d);
            const dayEnd = endOfDay(d);
            const { millis } = computeDailyDowntime(dayStart, dayEnd, offline);
            totalDownMs += millis;
          }
          const uptimePct = totalMs > 0 ? Math.max(0, 100 * (1 - (totalDownMs / totalMs))) : 100;
          nameCell.textContent = `${nvr.name || ''} (${nvr.ip}) — ${uptimePct.toFixed(1)}%`;
          row.appendChild(nameCell);
          for (let j = 0; j < days.length; j++) {
            const d = days[j];
            const dayStart = startOfDay(d);
            const dayEnd = endOfDay(d);
            const { millis, ratio } = computeDailyDowntime(dayStart, dayEnd, offline);
            const c = document.createElement('div');
            c.className = 'cell day';
            c.title = `${d.toLocaleDateString()} — downtime ${(millis/3600000).toFixed(2)}h`;
            const outlineColor = millis === 0 ? '#d1d5db' : colorForRatio(ratio);
            const outlineW = millis === 0 ? 2 : 3;
            c.style.boxShadow = `inset 0 0 0 ${outlineW}px ${outlineColor}`;
            c.addEventListener('mouseenter', () => {
              nameCell.classList.add('active');
              headerDayCells[j].classList.add('active');
            });
            c.addEventListener('mouseleave', () => {
              nameCell.classList.remove('active');
              headerDayCells[j].classList.remove('active');
            });
            c.addEventListener('click', () => showDayDetail(nvr, dayStart, dayEnd, offline));
            row.appendChild(c);
          }
          wrapper.appendChild(row);
        }
        container.appendChild(wrapper);
      }

      function showDayDetail(nvr, dayStart, dayEnd, offlineIntervals) {
        const overlay = document.createElement('div');
        overlay.className = 'modal-overlay';
        const modal = document.createElement('div');
        modal.className = 'modal';
        const h = document.createElement('h3');
        h.textContent = `${nvr.name || ''} (${nvr.ip}) — ${dayStart.toLocaleDateString()}`;
        const list = document.createElement('div');
        const items = [];
        for (const iv of offlineIntervals) {
          const s = new Date(iv.start * 1000);
          const e = new Date((iv.end ?? Math.floor(Date.now()/1000)) * 1000);
          if (e < dayStart || s > dayEnd) continue;
          const start = new Date(Math.max(s.getTime(), dayStart.getTime()));
          const end = new Date(Math.min(e.getTime(), dayEnd.getTime()));
          const durMs = end - start;
          if (durMs <= 0) continue;
          items.push({ start, end, durMs });
        }
        items.sort((a,b) => a.start - b.start);
        if (items.length === 0) {
          list.textContent = 'No downtime intervals for this day.';
        } else {
          const ul = document.createElement('ul');
          for (const it of items) {
            const li = document.createElement('li');
            li.textContent = `${it.start.toLocaleTimeString()} ? ${it.end.toLocaleTimeString()}  (${(it.durMs/60000).toFixed(0)} min)`;
            ul.appendChild(li);
          }
          list.appendChild(ul);
        }
        const close = document.createElement('button');
        close.textContent = 'Close';
        close.addEventListener('click', () => { document.body.removeChild(overlay); });
        modal.appendChild(h);
        modal.appendChild(list);
        modal.appendChild(close);
        overlay.appendChild(modal);
        document.body.appendChild(overlay);
      }

      async function init() {
        const { from, to } = defaultRange();
        setRangeInputs(from, to);
        let nvrs = await loadNvrs();
        let events = await loadEvents(from, to);
        let days = buildDays(from, to);
        renderCalendar(nvrs, events, days);

        // Load button handler
        document.getElementById('load-range').addEventListener('click', async () => {
          const fromInput = document.getElementById('from-date').value;
          const toInput = document.getElementById('to-date').value;
          if (!fromInput || !toInput) return;
          const from = new Date(fromInput + 'T00:00:00');
          const to = new Date(toInput + 'T23:59:59');
          setRangeInputs(from, to);
          nvrs = await loadNvrs();
          events = await loadEvents(from, to);
          days = buildDays(from, to);
          renderCalendar(nvrs, events, days);
        });

        // Quick preset handlers
        document.querySelectorAll('.preset-btn').forEach(btn => {
          btn.addEventListener('click', async () => {
            document.querySelectorAll('.preset-btn').forEach(b => {
              b.classList.remove('active', 'border-blue-500', 'text-blue-600', 'bg-blue-50');
              b.style.borderColor = '';
              b.style.color = '';
              b.style.backgroundColor = '';
            });
            btn.classList.add('active');
            btn.style.borderColor = '#3b82f6';
            btn.style.color = '#2563eb';
            btn.style.backgroundColor = '#eff6ff';
            
            const daysCount = parseInt(btn.dataset.days, 10);
            const { from, to } = applyPreset(daysCount);
            setRangeInputs(from, to);
            nvrs = await loadNvrs();
            events = await loadEvents(from, to);
            days = buildDays(from, to);
            renderCalendar(nvrs, events, days);
          });
        });

        document.getElementById('nvr-filter').addEventListener('input', () => {
          renderCalendar(nvrs, events, days);
        });
      }

      window.addEventListener('DOMContentLoaded', init);
    </script>
